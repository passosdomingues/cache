/* EXEMPLO 1: APENAS SHOULD (O "Pelo Menos Um")
   A intenção é buscar artigos que falem de 'treap' OU 'binary search'.
   Como não há 'must', o Elastic obriga que pelo menos um 'should' seja verdade.
*/
GET /wikipedia/_search
{
  "query": {
    "bool": {
      "should": [
        { "match": { "content": "treap" } },
        { "match": { "content": "binary search" } }
      ]
    }
  }
}

/* EXEMPLO 2: BUSCA POR EXCLUSÃO (must_not)
   Intenção: Buscar 'square root', mas EXCLUIR resultados que falem 
   de 'functional' (talvez você queira apenas a raiz aritmética comum).
*/
GET /wikipedia/_search
{
  "query": {
    "bool": {
      "must": [
        { "match": { "content": "square root" } }
      ],
      "must_not": [
        { "match": { "content": "functional" } }
      ]
    }
  }
}

/* EXEMPLO 3: COMBO COMPLETO (Must + Should + Filter + Must_Not)
   Intenção: 
   - OBRIGATÓRIO: conter "the treap" e "randomized binary" (frase).
   - RELEVÂNCIA: Se tiver "tree" no título, aparece primeiro.
   - FILTRO: Criado apenas após o ano 2000 (não afeta score).
   - EXCLUSÃO: Não pode conter a palavra "deprecated".
*/
GET /wikipedia/_search
{
  "query": {
    "bool": {
      "must": [
        { "match": { "content": "the treap" } },
        { "match_phrase": { "content": "randomized binary" } }
      ],
      "should": [
        { "match": { "title": "tree" } }
      ],
      "filter": [
        { "range": { "dt_creation": { "gte": "2000-01-01" } } }
      ],
      "must_not": [
        { "match": { "content": "deprecated" } }
      ]
    }
  }
}
